-- ستون ردیابی نقاط ضعف رفع‌شده alter table public.evaluations add column resolved_weaknesses jsonb not null default '[]'::jsonb; -- تریگر محافظ: کاربر می‌تونه ردیف ارزیابی خودش رو آپدیت کنه (برای resolved_weaknesses)، -- ولی is_paid همیشه دست‌نخورده می‌مونه مگر اینکه service_role (یعنی n8n) خودش تغییرش بده. create or replace function protect_is_paid() returns trigger as $$ begin if auth.role() <> 'service_role' then new.is_paid := old.is_paid; end if; return new; end; $$ language plpgsql security definer; drop trigger if exists trg_protect_is_paid on public.evaluations; create trigger trg_protect_is_paid before update on public.evaluations for each row execute function protect_is_paid(); -- حالا که is_paid محافظت‌شده، اجازه UPDATE عمومی (فقط روی ردیف خودشون) امن هست create policy "Users can update their own evaluations" on public.evaluations for update using (auth.uid() = user_id) with check (auth.uid() = user_id);
